# Windows publish and push command lines, as agents on Windows send them (PowerShell, cmd).
# Read by tests/test_commands_windows.py: each one goes through commands.detect as the guard
# runs it on Windows, and must give exactly what is written after "=>".
#
#   <command line>  =>  <what>[ ; <what> ...]
#   <what> = <action> <tool> <folder> [<target> ...] [unsure]   or   nothing
#
# <folder> is where the command runs, relative to the agent's folder, with forward slashes
# (`.` is the agent's own); quote it when it has a space. "unsure" means agentprint says "could
# not check", never OK.

# npm and the other Node publishers, as Windows names them

npm.cmd publish  =>  publish npm .
NPM.CMD publish --access public  =>  publish npm .
npm.ps1 publish  =>  publish npm .

& npm publish  =>  publish npm .

pnpm.cmd -C packages\a publish  =>  publish pnpm packages/a
yarn.cmd npm publish  =>  publish yarn .
bun.exe publish  =>  publish bun .
npx.cmd changeset publish  =>  publish changeset .
.\node_modules\.bin\changeset.cmd publish  =>  publish changeset .
npx.cmd np  =>  publish np .

# powershell / pwsh, unwrapped like bash -c
powershell -Command "npm publish"  =>  publish npm .

powershell "npm publish"  =>  publish npm .

pwsh -c 'git push origin main'  =>  push git .
pwsh -NoLogo -NonInteractive -Command "Set-Location C:\pkg; git push"  =>  push git C:/pkg
pwsh -WorkingDirectory pkg -Command "npm publish"  =>  publish npm pkg
powershell -EncodedCommand bgBwAG0AIABwAHUAYgBsAGkAcwBoAA==  =>  publish npm .
pwsh -cwa 'npm $args[0]' publish  =>  publish npm . unsure

# Start-Process, Invoke-Expression, wsl
Start-Process npm -ArgumentList 'publish'  =>  publish npm .
Start-Process -FilePath npm.cmd -ArgumentList 'publish','--access','public' -Wait -NoNewWindow  =>  publish npm .
Start-Process git -ArgumentList push -WorkingDirectory C:\repo  =>  push git C:/repo
start npm publish  =>  publish npm .
saps npm.cmd 'publish', '--tag', 'next'  =>  publish npm .
Invoke-Expression "npm publish"  =>  publish npm .
iex 'git push'  =>  push git .
wsl npm publish  =>  publish npm .
wsl -e git push  =>  push git .

# PowerShell around the publish: assignments, redirections, blocks, subexpressions
$env:NPM_CONFIG_PROVENANCE = 'true'; npm publish  =>  publish npm .
$out = npm publish 2>&1  =>  publish npm .
$out=npm.cmd publish  =>  publish npm .
[void](npm publish)  =>  publish npm .
npm publish *> $null  =>  publish npm .
npm publish 2>$null | Out-Null  =>  publish npm .
npm publish > publish.log  =>  publish npm .
if ($?) { npm publish }  =>  publish npm .
if (Test-Path dist) { twine upload dist\* }  =>  publish twine . dist/*
try { npm publish } catch { Write-Host failed }  =>  publish npm .
Write-Host "published: $(npm publish)"  =>  publish npm .
npm test && npm publish  =>  publish npm .
npm test || exit 1; npm publish  =>  publish npm .
npm publish; git push --follow-tags  =>  publish npm . ; push git .
npm publish (Get-Item .\pkg)  =>  publish npm . $(...) unsure
twine upload (Get-ChildItem dist\*.whl)  =>  publish twine . $(...) unsure
& $npm publish  =>  publish $npm . unsure
foreach ($p in 'a','b') { Push-Location $p; npm publish; Pop-Location }  =>  publish npm . unsure
Get-ChildItem packages | ForEach-Object { Push-Location $_.FullName; npm publish; Pop-Location }  =>  publish npm . unsure
foreach ($p in 'a','b') { cd $p }; npm publish  =>  publish npm . unsure
npm publish "unclosed  =>  publish npm . unclosed unsure
